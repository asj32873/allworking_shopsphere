# Jenkins now builds, pushes, and deploys registry images. See ..\jenkins\README.md.
# These commands are only for a manual deployment after creating both secrets.
cd microservices
kubectl apply -f k8s\00-namespace.yaml

  --docker-server=docker.io --docker-username=<username> --docker-password=<access-token>
kubectl apply -k k8s
